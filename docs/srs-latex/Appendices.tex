\section{Main API routes}
\begin{table}[htbp]
\centering
\small
\begin{tabularx}{\textwidth}{>{\raggedright\arraybackslash}p{8.2cm}L}
\hline
\textbf{Route} & \textbf{Who and what} \\ \hline
\texttt{POST /api/auth/register}, \texttt{/login} & Anyone: create an account, log in \\
\texttt{GET/PUT /api/users/me} & Any user: profile, wallet, balance \\
\texttt{GET/POST /api/jobs} & Client posts; everyone browses open jobs \\
\texttt{POST /api/jobs/:id/proposals} & Freelancer sends a proposal \\
\texttt{POST /api/proposals/:id/accept} & Client accepts; escrow is deployed \\
\texttt{POST /api/jobs/:id/milestones} & Client adds a milestone \\
\texttt{POST /api/milestones/:id/fund} & Client funds \\
\texttt{POST /api/milestones/:id/submit} & Freelancer submits; risk check runs \\
\texttt{POST /api/milestones/:id/approve}, \texttt{/dispute} & Client decides \\
\texttt{POST /api/milestones/:id/appeal} & Freelancer appeals a flag \\
\texttt{GET /api/arbiter/queue} & Arbiter sees held and disputed milestones \\
\texttt{POST /api/arbiter/milestones/:id/}\\\texttt{release} or \texttt{refund} & Arbiter decides \\
\texttt{POST /api/arbiter/jobs/:id/pause}, \texttt{/unpause} & Arbiter emergency stop \\ \hline
\end{tabularx}
\caption{Main REST API routes.}
\label{tab:api}
\end{table}
